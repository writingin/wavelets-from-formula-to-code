import numpy as np

def haar_step(signal):
    """Single-level 1D Discrete Haar Wavelet Transform."""
    n = len(signal)
    approx = (signal[0:n:2] + signal[1:n:2]) / np.sqrt(2)
    detail = (signal[0:n:2] - signal[1:n:2]) / np.sqrt(2)
    return approx, detail
